type Service struct {
	Repository      *Repository
	SchoolService   *school.Service
	ResultService   *result.Service
	ExamService     *exam.Service
	ClassService    *class.Service
	UnitService     *unit.Service
	SequenceService *sequence.Service
	QuarterService  *quarter.Service
	SemesterService *semester.Service
}

func NewService(
	repository *Repository,
	schoolService *school.Service,
	resultService *result.Service,
	examService *exam.Service,
	classService *class.Service,
	unitService *unit.Service,
	sequenceService *sequence.Service,
	quarterService *quarter.Service,
	semesterService *semester.Service,
) *Service {
	return &Service{
		Repository:      repository,
		SchoolService:   schoolService,
		ResultService:   resultService,
		ExamService:     examService,
		ClassService:    classService,
		UnitService:     unitService,
		SequenceService: sequenceService,
		QuarterService:  quarterService,
		SemesterService: semesterService,
	}
}

const MODEL_NAME = "report"
const DEFAULT_ERROR_MESSAGE = "interact with report model"

func (service *Service) CreateEntry(
	ctxData *types.ContextData,
	request *data.ReportEntryRequest,
) (result *model.ReportEntry, errCode int, err error) {
	// Check school
	newRequest := *request
	if ctxData.Jwt.SchoolID > 0 {
		newRequest.SchoolID = ctxData.Jwt.SchoolID
	}
	if newRequest.SchoolID < 1 {
		errCode = http.StatusBadRequest
		err = constants.Http400BadRequestErrorMessage()
		return
	}

	// Find school
	foundSchool, err := service.SchoolService.Repository.GetByID(newRequest.SchoolID)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	if foundSchool == nil || foundSchool.ID < 1 {
		errCode = http.StatusNotFound
		err = constants.Http404ErrorMessage(MODEL_NAME)
		return
	}

	// Find report configuration
	foundReportConfig, err := service.Repository.GetReportConfigBySchoolID(newRequest.SchoolID)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	if foundReportConfig == nil || foundReportConfig.ID < 1 {
		errCode = http.StatusNotFound
		err = constants.Http404ErrorMessage(MODEL_NAME)
		return
	}

	// Find all results
	resultRequest := &dataResult.GetAllRequest{}
	resultRequest.SchoolID = newRequest.SchoolID
	resultRequest.YearID = newRequest.YearID
	if foundSchool.Type == constants.SCHOOL_TYPE_HIGHSCHOOL {
		resultRequest.ClassID = newRequest.ClassID
		if newRequest.PeriodType == constants.REPORT_PERIOD_TYPE_SEQUENCE {
			resultRequest.SequenceID = newRequest.SequenceID
		}
		if newRequest.PeriodType == constants.REPORT_PERIOD_TYPE_QUARTER {
			resultRequest.QuarterID = newRequest.QuarterID
		}
	} else if foundSchool.Type == constants.SCHOOL_TYPE_UNIVERSITY {
		resultRequest.LevelDomainID = newRequest.LevelDomainID
		if newRequest.PeriodType == constants.REPORT_PERIOD_TYPE_SEMESTER {
			resultRequest.SemesterID = newRequest.SemesterID
		}
	}
	foundResults, err := service.ResultService.Repository.GetAll(nil, nil, resultRequest)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	if foundResults == nil || len(foundResults) < 1 {
		errCode = http.StatusNotFound
		err = constants.Http404ErrorMessage(MODEL_NAME)
		return
	}

	// Group results by student and sequence and calculate final grade
	finalEntries := map[string]*model.ReportEntry{}
	finalNotes := map[string]float64{}
	for _, res := range foundResults {
		if res.Exam == nil || res.Exam.Notation == 0 || res.Exam.Percentage == 0 {
			continue // skip division by zero
		}

		// Set the map key
		mapKey := fmt.Sprintf("%sseq%d", res.StudentID, res.Exam.SequenceID)

		// Normalize score
		normalized := (res.Score / res.Exam.Notation) * float64(res.Exam.Percentage)
		finalNotes[mapKey] += normalized

		// Create entry
		foundEntry := finalEntries[mapKey]
		if foundEntry == nil {
			tempNewEntry := &model.ReportEntry{
				StudentID:      res.StudentID,
				SchoolID:       res.SchoolID,
				YearID:         res.Exam.YearID,
				ClassSubjectID: res.Exam.ClassSubjectID,
				UnitID:         res.Exam.UnitID,
				SequenceID:     res.Exam.SequenceID,

				Notation: foundReportConfig.NotationReport,
			}
			if res.Exam.Unit != nil && res.Exam.Unit.Credit > 0 {
				tempNewEntry.CoefficientCredit = res.Exam.Unit.Credit
			} else if res.Exam.ClassSubject != nil && res.Exam.ClassSubject.Coefficient > 0 {
				tempNewEntry.CoefficientCredit = res.Exam.ClassSubject.Coefficient
			}
			finalEntries[mapKey] = tempNewEntry
		}

		// Update entry
		if !foundEntry.IsRetry {
			foundEntry.IsRetry = res.Exam.IsRetry
		}
		foundEntry.RetryCount += res.Exam.RetryCount
		if res.Exam.RetryCount > 0 && res.Exam.Type != nil {
			if len(foundEntry.RetryDetails) > 0 {
				foundEntry.RetryDetails = fmt.Sprintf("%s, %s(%d)", foundEntry.RetryDetails, res.Exam.Type.Name, res.Exam.RetryCount)
			} else {
				foundEntry.RetryDetails = fmt.Sprintf("%s(%d)", res.Exam.Type.Name, res.Exam.RetryCount)
			}
		}
	}

	// Normalize the report entry score with report config
	if foundReportConfig.NotationReport > 0 {
		for mapKey, score := range finalNotes {
			tempFinalScore := (score / 100) * float64(foundReportConfig.NotationReport)
			foundEntry := finalEntries[mapKey]
			if foundEntry != nil && foundEntry.StudentID > 0 {
				foundEntry.Score = tempFinalScore
			}
		}
	}

	// Update the entry grade
	reportGradeRequest := &data.GetAllReportGradeRequest{}
	reportGradeRequest.SchoolID = newRequest.SchoolID
	reportGradeRequest.Type = constants.REPORT_GRADE_TYPE_REPORT
	foundReportGrades, err := service.Repository.GetAllReportGrade(nil, nil, reportGradeRequest)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	if len(foundReportGrades) > 0 {
		for _, entry := range finalEntries {
			if entry == nil {
				continue
			}

			for _, grade := range foundReportGrades {
				if grade.Minimum > grade.Maximum {
					continue
				}

				minOK := (grade.IncludeMinimum && entry.Score >= grade.Minimum) || (!grade.IncludeMinimum && entry.Score > grade.Minimum)
				maxOK := (grade.IncludeMaximum && entry.Score <= grade.Maximum) || (!grade.IncludeMaximum && entry.Score < grade.Maximum)

				if minOK && maxOK {
					entry.GradeName = grade.Name
					entry.GradeDescription = grade.Description
					break
				}
			}
		}
	}

	// Get the total coefficient credit
	finalTotalCoefficientCredit := 0
	if foundSchool.Type == constants.SCHOOL_TYPE_HIGHSCHOOL {
		classSubjectRequest := &dataClass.GetAllClassSubjectRequest{}
		classSubjectRequest.SchoolID = newRequest.SchoolID
		classSubjectRequest.ClassID = newRequest.ClassID
		foundClassSubjects, errFound := service.ClassService.Repository.GetAllClassSubject(nil, nil, classSubjectRequest)
		if errFound != nil {
			errCode = http.StatusInternalServerError
			err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
			return
		}
		for _, classSubject := range foundClassSubjects {
			if classSubject.Coefficient > 0 && classSubject.IsValid {
				finalTotalCoefficientCredit += classSubject.Coefficient
			}
		}
	} else if foundSchool.Type == constants.SCHOOL_TYPE_UNIVERSITY {
		unitRequest := &dataUnit.GetAllRequest{}
		unitRequest.SchoolID = newRequest.SchoolID
		unitRequest.LevelDomainID = newRequest.LevelDomainID
		unitRequest.SemesterID = newRequest.SemesterID
		foundUnits, errFound := service.UnitService.Repository.GetAll(nil, nil, unitRequest)
		if errFound != nil {
			errCode = http.StatusInternalServerError
			err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
			return
		}
		for _, unit := range foundUnits {
			if unit.Credit > 0 && unit.IsValid {
				finalTotalCoefficientCredit += unit.Credit
			}
		}
	}

	// Get the period name
	periodName := ""
	if foundSchool.Type == constants.SCHOOL_TYPE_HIGHSCHOOL {
		resultRequest.ClassID = newRequest.ClassID
		if newRequest.PeriodType == constants.REPORT_PERIOD_TYPE_SEQUENCE {
			foundSequence, errFound := service.SequenceService.Repository.GetByIDSchoolID(newRequest.SequenceID, newRequest.SchoolID)
			if errFound != nil {
				errCode = http.StatusInternalServerError
				err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
				return
			}
			if foundSequence != nil && foundSequence.ID > 0 {
				periodName = foundSequence.Name
			}
		}
		if newRequest.PeriodType == constants.REPORT_PERIOD_TYPE_QUARTER {
			foundQuarter, errFound := service.QuarterService.Repository.GetByIDSchoolID(newRequest.QuarterID, newRequest.SchoolID)
			if errFound != nil {
				errCode = http.StatusInternalServerError
				err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
				return
			}
			if foundQuarter != nil && foundQuarter.ID > 0 {
				periodName = foundQuarter.Name
			}
		}
	} else if foundSchool.Type == constants.SCHOOL_TYPE_UNIVERSITY {
		if newRequest.PeriodType == constants.REPORT_PERIOD_TYPE_SEMESTER {
			foundSemester, errFound := service.SemesterService.Repository.GetByIDSchoolID(newRequest.SemesterID, newRequest.SchoolID)
			if errFound != nil {
				errCode = http.StatusInternalServerError
				err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
				return
			}
			if foundSemester != nil && foundSemester.ID > 0 {
				periodName = foundSemester.Name
			}
		}
	}

	// Group average by student
	finalAverages := map[string]*model.ReportAverage{}
	finalAveragesScore := map[string]float64{}
	for _, entry := range finalEntries {
		if entry == nil {
			continue
		}

		// Set the map key
		mapKey := fmt.Sprintf("%s", entry.StudentID)

		// Create new average
		foundAvg := finalAverages[mapKey]
		if foundAvg == nil {
			foundAvg = &model.ReportAverage{
				StudentID:     entry.StudentID,
				SchoolID:      entry.SchoolID,
				YearID:        entry.YearID,
				ClassID:       newRequest.ClassID,
				LevelDomainID: newRequest.LevelDomainID,

				PeriodType:             newRequest.PeriodType,
				PeriodName:             periodName,
				Notation:               foundReportConfig.NotationAverage,
				TotalCreditCoefficient: finalTotalCoefficientCredit,
				IsSuccessful:           true,
			}
		}

		// Update the total credit
		foundAvg.TotalCreditCoefficient += entry.CoefficientCredit
		if foundReportConfig.ReportGradeToFail != nil && entry.GradeName == foundReportConfig.ReportGradeToFail.Name {
			foundAvg.IsSuccessful = false
		}
		// Update the score = sum of (entries/notation report) * coefficient credit)

		// Calculate and normalize the score
		var scoreWithCoefficientCredit float64 = (entry.Score / entry.Notation) * float64(entry.CoefficientCredit)
	}

	// Update and normalize the score
	// Score                      float64 `gorm:"default:null"`
	// IsSuccessful               bool    `gorm:"default:null"`
	// ValidatedCreditCoefficient int     `gorm:"default:null"`

	// Update the average grade
	avarageRequest := &data.GetAllReportGradeRequest{}
	avarageRequest.SchoolID = newRequest.SchoolID
	avarageRequest.Type = constants.REPORT_GRADE_TYPE_AVERAGE
	foundAverageGrades, err := service.Repository.GetAllReportGrade(nil, nil, avarageRequest)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	if len(foundAverageGrades) > 0 {
		for _, avg := range finalAverages {
			if avg == nil {
				continue
			}

			for _, grade := range foundAverageGrades {
				if grade.Minimum > grade.Maximum {
					continue
				}

				minOK := (grade.IncludeMinimum && avg.Score >= grade.Minimum) || (!grade.IncludeMinimum && avg.Score > grade.Minimum)
				maxOK := (grade.IncludeMaximum && avg.Score <= grade.Maximum) || (!grade.IncludeMaximum && avg.Score < grade.Maximum)

				if minOK && maxOK {
					avg.GradeName = grade.Name
					avg.GradeDescription = grade.Description
					break
				}
			}
		}
	}

	// Create entries
	for studentID, score := range finalNotes {
		entries = append(entries, entry)
		_, _ = service.Repository.CreateReportEntry(entry)
		if err != nil {
			utils.Logger.Info("Error creating report entry", zap.Error(err))
		}
	}
	// Create averages
	for studentID, score := range finalNotes {
		entries = append(entries, entry)
		_, _ = service.Repository.CreateReportEntry(entry)
		if err != nil {
			utils.Logger.Info("Error creating report entry", zap.Error(err))
		}
	}
	// Create table
	finalTable := &model.ReportTable{
		SchoolID:      newRequest.SchoolID,
		YearID:        newRequest.YearID,
		ClassID:       newRequest.ClassID,
		LevelDomainID: newRequest.LevelDomainID,

		PeriodType: newRequest.PeriodType,
		PeriodName: periodName,
		Status:     constants.REPORT_TABLE_STATUS_GENERATED,
	}
	_, err = service.Repository.CreateReportTable(finalTable)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	return
}

func (service *Service) CreateGrade(
	ctxData *types.ContextData,
	request *data.ReportGradeRequest,
) (result *model.ReportGrade, errCode int, err error) {
	// Check school
	newRequest := *request
	if ctxData.Jwt.SchoolID > 0 {
		newRequest.SchoolID = ctxData.Jwt.SchoolID
	}

	// Format
	item := &model.ReportGrade{
		SchoolID: newRequest.SchoolID,

		Type:           newRequest.Type,
		Name:           newRequest.Name,
		Description:    newRequest.Description,
		Minimum:        newRequest.Minimum,
		Maximum:        newRequest.Maximum,
		IncludeMinimum: newRequest.IncludeMinimum,
		IncludeMaximum: newRequest.IncludeMaximum,
	}

	// Check unique
	foundUnique, err := service.Repository.GetUniqueObjectReportGrade(item)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	if service.Repository.AreSameUniqueObjectsReportGrade(foundUnique, item) {
		errCode = http.StatusFound
		err = constants.Http302ErrorMessage(MODEL_NAME)
		return
	}

	// Create
	result, err = service.Repository.CreateReportGrade(item)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	return
}

func (service *Service) UpdateGrade(
	ctxData *types.ContextData,
	id int64,
	request *data.ReportGradeRequest,
) (result *model.ReportGrade, errCode int, err error) {
	// Check school
	newRequest := *request
	if ctxData.Jwt.SchoolID > 0 {
		newRequest.SchoolID = ctxData.Jwt.SchoolID
	}

	// Check if the item exists
	var foundItem *model.ReportGrade
	if ctxData.User.Feature != constants.FeatureAdmin {
		foundItem, err = service.Repository.GetReportGradeByIDSchoolID(id, newRequest.SchoolID)
	} else {
		foundItem, err = service.Repository.GetReportGradeByID(id)
	}
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	if foundItem == nil || foundItem.ID < 1 {
		errCode = http.StatusNotFound
		err = constants.Http404ErrorMessage(MODEL_NAME)
		return
	}

	// Format request
	item := &model.ReportGrade{
		SchoolID: newRequest.SchoolID,

		Type:           newRequest.Type,
		Name:           newRequest.Name,
		Description:    newRequest.Description,
		Minimum:        newRequest.Minimum,
		Maximum:        newRequest.Maximum,
		IncludeMinimum: newRequest.IncludeMinimum,
		IncludeMaximum: newRequest.IncludeMaximum,
	}

	// Check unique
	foundUnique, err := service.Repository.GetUniqueObjectReportGrade(item)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	if service.Repository.AreSameUniqueObjectsReportGrade(foundUnique, item) &&
		!service.Repository.AreSameUniqueObjectsReportGrade(foundUnique, foundItem) {
		errCode = http.StatusFound
		err = constants.Http302ErrorMessage(MODEL_NAME)
		return
	}

	// Update
	result, err = service.Repository.UpdateReportGradeByID(id, item)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		return
	}
	return
}

func (service *Service) DeleteGrade(

func (service *Service) GetAllTable(
	ctxData *types.ContextData,
	filter *types.Filter,
	pagination *types.Pagination,
	request *data.GetAllReportTableRequest,
) (result []model.ReportTable, errCode int, err error) {
	// Check school
	newRequest := *request
	if ctxData.Jwt.SchoolID > 0 {
		newRequest.SchoolID = ctxData.Jwt.SchoolID
	}

	// Get school
	if newRequest.SchoolID > 0 {
		foundSchool, errFound := service.SchoolService.Repository.GetByID(newRequest.SchoolID)
		if errFound != nil {
			errCode = http.StatusInternalServerError
			err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
		}
		if foundSchool == nil || foundSchool.ID < 1 {
			errCode = http.StatusNotFound
			err = constants.Http404ErrorMessage(MODEL_NAME)
			return
		}

		// Get
		switch foundSchool.Type {
		case constants.SCHOOL_TYPE_HIGHSCHOOL:
			result, err = service.Repository.GetAllReportTableHighschool(filter, pagination, &newRequest)
			if err != nil {
				errCode = http.StatusInternalServerError
				err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
			}
		case constants.SCHOOL_TYPE_UNIVERSITY:
			result, err = service.Repository.GetAllReportTableUniversity(filter, pagination, &newRequest)
			if err != nil {
				errCode = http.StatusInternalServerError
				err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
			}
		}
		return
	}

	// Get
	result, err = service.Repository.GetAllReportTable(filter, pagination, &newRequest)
	if err != nil {
		errCode = http.StatusInternalServerError
		err = constants.Http500ErrorMessage(DEFAULT_ERROR_MESSAGE)
	}
	return
}
